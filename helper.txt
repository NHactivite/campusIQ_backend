import { Request, Response } from "express";
import { isValidObjectId, Types } from "mongoose";
import Event, { IMedia, EVENT_CATEGORIES } from "../models/Event";
import { uploadEventFiles, deleteMedia } from "../utils/cloudinary";
import { asyncHandler, ApiError, slugify, toArray } from "../utils/helpers";
import { cached, cacheGet, cacheSet, invalidate, publicCache, noStore } from "../utils/cache";

type Files = { [field: string]: Express.Multer.File[] } | undefined;
const folderFor = (slug: string) => `iic/events/${slug}`;
const allMedia = (e: { images: IMedia[]; videos: IMedia[]; documents: IMedia[] }) =>
  [...e.images, ...e.videos, ...e.documents];

const pickFields = (b: Request["body"]) => {
  const out: Record<string, any> = {};
  for (const k of ["title", "description", "category", "venue", "organiser", "outcome"]) if (b[k] !== undefined) out[k] = b[k];
  if (b.eventDate) out.eventDate = new Date(b.eventDate);
  if (b.endDate) out.endDate = new Date(b.endDate);
  if (b.participants !== undefined && b.participants !== "") out.participants = Number(b.participants);
  if (b.isPublished !== undefined) out.isPublished = b.isPublished === true || b.isPublished === "true";
  return out;
};

// POST /api/events  (admin, multipart/form-data)
export const createEvent = asyncHandler(async (req, res) => {
  const fields = pickFields(req.body);
  if (!fields.title) throw new ApiError(400, "Title is required");

  const event = new Event({ ...fields, slug: slugify(fields.title), createdBy: req.user?.id });
  const uploaded = await uploadEventFiles(req.files as Files, folderFor(event.title));
  event.images = uploaded.images;
  event.videos = uploaded.videos;
  event.documents = uploaded.documents;

  try {
    await event.save();
  } catch (err) {
    await deleteMedia(allMedia(uploaded)); // don't leave orphan files in Cloudinary
    throw err;
  }
  await invalidate("events");
  res.status(201).json({ success: true, data: event });
});

// GET /api/events/admin/all  (admin: includes drafts)
// extra admin filter: ?published=true|false
export const getAdminEvents = asyncHandler(async (req, res) => buildList(req, res, false));

async function buildList(req: Request, res: Response, publicOnly: boolean) {
  const { category, search, status, published } = req.query as Record<string, string | undefined>;
  const page = Math.max(1, Number(req.query.page) || 1);
  const limit = Math.min(50, Math.max(1, Number(req.query.limit) || 12));

  // only accept known values, so junk query strings can't create endless cache keys
  const validCategory = category && (EVENT_CATEGORIES as readonly string[]).includes(category) ? category : undefined;
  const validStatus = status === "upcoming" || status === "past" ? status : undefined;

  const filter: Record<string, any> = {};
  if (publicOnly) filter.isPublished = true;
  else if (published === "true") filter.isPublished = true;    // admin only
  else if (published === "false") filter.isPublished = false;  // admin only
  if (validCategory) filter.category = validCategory;
  if (search) filter.title = { $regex: search.replace(/[.*+?^${}()|[\]\\]/g, "\\$&"), $options: "i" };
  if (validStatus === "upcoming") filter.eventDate = { $gte: new Date() };
  if (validStatus === "past") filter.eventDate = { $lt: new Date() };

  const load = async () => {
    const [items, total] = await Promise.all([
      Event.find(filter).sort({ eventDate: -1 }).skip((page - 1) * limit).limit(limit),
      Event.countDocuments(filter),
    ]);
    return { success: true, data: items, pagination: { page, limit, total, pages: Math.ceil(total / limit) } };
  };

  // Only the public list is cached. Searches are skipped (every query is different),
  // and the admin list is never cached so admins always see their latest edits.
  if (publicOnly && !search) {
    publicCache(res, 30);
    const key = `list:${validCategory ?? ""}:${validStatus ?? ""}:${page}:${limit}`;
    return res.json(await cached("events", key, 120, load));
  }
  noStore(res);
  res.json(await load());
}

// GET /api/events/admin/stats  (admin) – exact counts, however many events exist
export const getEventStats = asyncHandler(async (_req, res) => {
  const now = new Date();
  const [total, upcoming, drafts, byCategory, recent] = await Promise.all([
    Event.countDocuments(),
    Event.countDocuments({ eventDate: { $gte: now } }),
    Event.countDocuments({ isPublished: false }),
    Event.aggregate([{ $group: { _id: "$category", count: { $sum: 1 } } }]),
    Event.find().sort({ eventDate: -1 }).limit(5),
  ]);

  noStore(res);
  res.json({
    success: true,
    data: {
      total,
      upcoming,
      past: total - upcoming,
      drafts,
      byCategory: Object.fromEntries(byCategory.map((c) => [c._id, c.count])),
      recent,
    },
  });
});

// GET /api/events/:idOrSlug  (public)
export const getEvent = asyncHandler(async (req, res) => {
  const key = req.params.idOrSlug as string;
  const ckey = `one:${key.slice(0, 200)}`;

  const hit = await cacheGet<{ success: boolean; data: unknown }>("events", ckey);
  if (hit) {
    publicCache(res, 30);
    return res.json(hit);
  }

  const event = await Event.findOne(isValidObjectId(key) ? { _id: key } : { slug: key });
  if (!event || (!event.isPublished && req.user?.role !== "admin")) throw new ApiError(404, "Event not found");

  const body = { success: true, data: event };
  if (event.isPublished) {
    // only published events are stored, so a draft (visible to admins) never leaks to the public
    await cacheSet("events", ckey, body, 300);
    publicCache(res, 30);
  } else {
    noStore(res);
  }
  res.json(body);
});

// PATCH /api/events/:id  (admin, multipart) – edit fields, add files, remove files via `removeMedia`
export const updateEvent = asyncHandler(async (req, res) => {
  const event = await Event.findById(req.params.id);
  if (!event) throw new ApiError(404, "Event not found");

  Object.assign(event, pickFields(req.body));

  // remove selected media (ids of the media subdocuments)
  const removeIds = new Set(toArray(req.body.removeMedia));
  const removed: IMedia[] = [];
  for (const key of ["images", "videos", "documents"] as const) {
    const keep: IMedia[] = [];
    for (const m of event[key]) (removeIds.has(String(m._id)) ? removed : keep).push(m);
    event[key] = keep as any;
  }

  // add newly uploaded media
  const uploaded = await uploadEventFiles(req.files as Files, folderFor(event.slug));
  event.images.push(...(uploaded.images as any));
  event.videos.push(...(uploaded.videos as any));
  event.documents.push(...(uploaded.documents as any));

  try {
    await event.save();
  } catch (err) {
    await deleteMedia(allMedia(uploaded));
    throw err;
  }
  await deleteMedia(removed); // only after the DB save succeeded
  await invalidate("events");
  res.json({ success: true, data: event });
});

// DELETE /api/events/:id  (admin)
export const deleteEvent = asyncHandler(async (req, res) => {
  const event = await Event.findByIdAndDelete(req.params.id);
  if (!event) throw new ApiError(404, "Event not found");
  await deleteMedia(allMedia(event));
  await invalidate("events");
  res.json({ success: true, message: "Event deleted" });
});

// DELETE /api/events/:id/media/:mediaId  (admin)
export const deleteEventMedia = asyncHandler(async (req, res) => {
  const event = await Event.findById(req.params.id);
  if (!event) throw new ApiError(404, "Event not found");

  const mediaId = req.params.mediaId as string;
  let removed: IMedia | undefined;
  for (const key of ["images", "videos", "documents"] as const) {
    const found = event[key].find((m) => String(m._id) === mediaId);
    if (found) { removed = found; event[key] = event[key].filter((m) => String(m._id) !== mediaId) as any; break; }
  }
  if (!removed) throw new ApiError(404, "Media not found");

  await event.save();
  await deleteMedia([removed]);
  await invalidate("events");
  res.json({ success: true, data: event });
});

import { Request } from "express";
import { isValidObjectId } from "mongoose";
import Feature, { FEATURE_SECTIONS, FeatureSection } from "../models/Feature";
import { asyncHandler, ApiError, toArray } from "../utils/helpers";
import { cached, invalidate, publicCache, noStore } from "../utils/cache";

const pickFields = (b: Request["body"]) => {
  const out: Record<string, any> = {};
  for (const k of ["title", "text", "icon", "section"]) if (b[k] !== undefined) out[k] = b[k];
  if (b.tags !== undefined) out.tags = toArray(b.tags).map((t) => t.trim()).filter(Boolean);
  if (b.order !== undefined && b.order !== "") out.order = Number(b.order);
  if (b.isPublished !== undefined) out.isPublished = b.isPublished === true || b.isPublished === "true";
  return out;
};

// validates AND narrows the type from string to "rdc" | "iic"
const parseSection = (value: unknown, fallback?: FeatureSection): FeatureSection | undefined => {
  if (value === undefined || value === "") return fallback;
  if (!(FEATURE_SECTIONS as readonly string[]).includes(value as string))
    throw new ApiError(400, `section must be one of: ${FEATURE_SECTIONS.join(", ")}`);
  return value as FeatureSection;
};

    success: true,
    data: await Feature.find({ section, isPublished: true }).sort({ order: 1, createdAt: 1 }),
  }));
  res.json(body);
});

// PATCH /api/features/reorder   (admin)  body: { items: [{ id, order }, ...] }
export const reorderFeatures = asyncHandler(async (req, res) => {
  const items = req.body.items as { id: string; order: number }[];
  if (!Array.isArray(items) || items.length === 0) throw new ApiError(400, "items array is required");
  if (items.some((i) => !isValidObjectId(i.id) || typeof i.order !== "number"))
    throw new ApiError(400, "Each item needs a valid id and numeric order");

  await Feature.bulkWrite(
    items.map((i) => ({ updateOne: { filter: { _id: i.id }, update: { $set: { order: i.order } } } }))
  );
  await invalidate("features");
  res.json({ success: true, message: "Order updated" });
});

import { Request, Response, NextFunction } from "express";
import { getAuth, clerkClient } from "@clerk/express";
import { ApiError } from "../utils/helpers";

// remembers the role check for 60 s, so each request doesn't call Clerk
const cache = new Map<string, { admin: boolean; at: number }>();
const TTL = 60_000;

const isAdminUser = async (userId: string, claims: any): Promise<boolean> => {
  if (claims?.metadata?.role) return claims.metadata.role === "admin"; // only if you customised the session token

  const hit = cache.get(userId);
  if (hit && Date.now() - hit.at < TTL) return hit.admin;

  const user = await clerkClient.users.getUser(userId);
  const admin = (user.publicMetadata as { role?: string })?.role === "admin";
  cache.set(userId, { admin, at: Date.now() });
  return admin;
};

export const requireAdmin = async (req: Request, _res: Response, next: NextFunction) => {
  try {
    const { userId, sessionClaims } = getAuth(req);
    if (!userId) throw new ApiError(401, "Not authenticated");
    if (!(await isAdminUser(userId, sessionClaims))) throw new ApiError(403, "Admin access required");
    req.user = { id: userId, role: "admin" };
    next();
  } catch (err) {
    next(err);
  }
};

// never blocks: sets req.user only when the caller is a signed-in admin (used on public routes)
export const optionalAdmin = async (req: Request, _res: Response, next: NextFunction) => {
  try {
    const { userId, sessionClaims } = getAuth(req);
    if (userId && (await isAdminUser(userId, sessionClaims))) req.user = { id: userId, role: "admin" };
  } catch {
    /* treat as public */
  }
  next();
};



import { Request, Response, NextFunction } from "express";
import multer from "multer";
import { ApiError } from "../utils/helpers";

export const notFound = (_req: Request, res: Response) =>
  res.status(404).json({ success: false, message: "Route not found" });

export const errorHandler = (err: any, _req: Request, res: Response, _next: NextFunction) => {
  let status = 500;
  let message = err.message || "Server error";

  if (err instanceof ApiError) status = err.status;
  else if (err instanceof multer.MulterError) status = 400;
  else if (err.name === "ValidationError") { status = 400; message = Object.values<any>(err.errors).map((e) => e.message).join(", "); }
  else if (err.name === "CastError") { status = 400; message = "Invalid id"; }

  if (status === 500) console.error(err);
  res.status(status).json({ success: false, message });
};


import { Request, Response, NextFunction } from "express";
import { apiLimiter, writeLimiter, Limiter } from "../utils/ratelimit.js";

const make =
  (limiter: Limiter, keyOf: (req: Request) => string) =>
  async (req: Request, res: Response, next: NextFunction) => {
    if (req.method === "OPTIONS") return next();

    try {
      const { success, limit, remaining, reset } = await limiter.limit(keyOf(req));
      res.setHeader("X-RateLimit-Limit", String(limit));
      res.setHeader("X-RateLimit-Remaining", String(remaining));
      res.setHeader("X-RateLimit-Reset", String(Math.ceil(reset / 1000)));

      if (!success) {
        res.setHeader("Retry-After", String(Math.max(1, Math.ceil((reset - Date.now()) / 1000))));
        return res.status(429).json({ success: false, message: "Too many requests. Please try again later." });
      }
    } catch (err) {
      console.error("Rate limit error:", err); // fail open: a Redis hiccup must not lock everyone out
    }
    next();
  };

export const rateLimitMiddleware = make(apiLimiter, (req) => req.ip ?? "unknown");

import { Response } from "express";
import { redis, PREFIX } from "./redis.js";

/**
 * Two layers, both optional:
 *   L1  a small in-memory map inside this server instance
 *   L2  Upstash Redis, shared by every instance
 *
 * Invalidation uses a version number per namespace ("events", "features"...).
 * The version is part of every cache key, so one INCR makes all old keys
 * unreachable at once (no SCAN, no key lists). They simply expire on their own.
 *
 * Everything fails open: if Redis is missing or erroring, requests just go to
 * the database, nothing breaks.
 */

const VERSION_TTL_MS = 3_000; // how long an instance trusts its copy of a version
const L1_TTL_MS = 60_000;
const L1_MAX = 500;

const versions = new Map<string, { v: number; exp: number }>();
const l1 = new Map<string, { value: unknown; exp: number }>();
const inflight = new Map<string, Promise<unknown>>();

async function version(ns: string): Promise<string | null> {
  if (!redis) return null;
  const hit = versions.get(ns);
  if (hit && hit.exp > Date.now()) return String(hit.v);
  try {
    const v = Number(await redis.get<number>(`${PREFIX}:ver:${ns}`)) || 0;
    versions.set(ns, { v, exp: Date.now() + VERSION_TTL_MS });
    return String(v);
  } catch (err) {
    console.error("[cache] version read failed:", err);
    return null; // caller skips the cache
  }
}

const fullKey = (ns: string, v: string, key: string) => `${PREFIX}:c:${ns}:${v}:${key}`;

function remember(k: string, value: unknown) {
  if (l1.size >= L1_MAX) l1.delete(l1.keys().next().value as string); // drop the oldest
  l1.set(k, { value, exp: Date.now() + L1_TTL_MS });
}

export async function cacheGet<T>(ns: string, key: string): Promise<T | null> {
  const v = await version(ns);
  if (v === null) return null;
  const k = fullKey(ns, v, key);

  const mem = l1.get(k);
  if (mem && mem.exp > Date.now()) return mem.value as T;

  try {
    const value = await redis!.get<T>(k);
    if (value !== null && value !== undefined) {
      remember(k, value);
      return value;
    }
  } catch (err) {
    console.error("[cache] read failed:", err);
  }
  return null;
}

export async function cacheSet(ns: string, key: string, value: unknown, ttlSec: number) {
  const v = await version(ns);
  if (v === null) return;
  const k = fullKey(ns, v, key);
  const plain = JSON.parse(JSON.stringify(value)); // mongoose documents -> plain JSON
  remember(k, plain);
  try {
    await redis!.set(k, plain, { ex: ttlSec });
  } catch (err) {
    console.error("[cache] write failed:", err);
  }
}

/** Get from cache, or run `loader` and store its result. Concurrent misses share one loader run. */
export async function cached<T>(ns: string, key: string, ttlSec: number, loader: () => Promise<T>): Promise<T> {
  const hit = await cacheGet<T>(ns, key);
  if (hit !== null) return hit;

  const id = `${ns}:${key}`;
  const pending = inflight.get(id) as Promise<T> | undefined;
  if (pending) return pending;

  const run = (async () => {
    const value = await loader();
    await cacheSet(ns, key, value, ttlSec);
    return value;
  })().finally(() => inflight.delete(id));
  inflight.set(id, run);
  return run;
}

/** Call after any write. Bumps the version so every cached entry in the namespace is dropped. */
export async function invalidate(...namespaces: string[]) {
  l1.clear();
  if (!redis) return;
  await Promise.all(
    namespaces.map(async (ns) => {
      versions.delete(ns);
      try {
        await redis!.incr(`${PREFIX}:ver:${ns}`);
      } catch (err) {
        console.error("[cache] invalidate failed (entries expire on their own):", err);
      }
    }),
  );
}

export type UploadedFiles = { images: IMedia[]; videos: IMedia[]; documents: IMedia[] };

// Uploads every file on the request and returns the media records
export async function uploadEventFiles(
  files: { [field: string]: Express.Multer.File[] } | undefined,
  folder: string
): Promise<UploadedFiles> {
  const out: UploadedFiles = { images: [], videos: [], documents: [] };
  if (!files) return out;

  const plan: [keyof UploadedFiles, ResourceType][] = [
    ["images", "image"],
    ["videos", "video"],
    ["documents", "raw"],
  ];
  try {
    for (const [field, type] of plan) {
      const list = files[field] ?? [];
      const results = await Promise.all(list.map((f) => uploadBuffer(f, folder, type)));
      results.forEach((r, i) => out[field].push(toMedia(r, list[i], type)));
    }
    return out;
  } catch (err) {
    await deleteMedia([...out.images, ...out.videos, ...out.documents]); // roll back partial uploads
    throw err;
  }
}

export const deleteMedia = async (items: IMedia[]) => {
  await Promise.allSettled(items.map((m) => destroyAsset(m.publicId, m.resourceType)));
};

import { Ratelimit } from "@upstash/ratelimit";
import { redis, PREFIX } from "./redis.js";

type Result = { success: boolean; limit: number; remaining: number; reset: number };
export type Limiter = { limit: (key: string) => Promise<Result> };

// per-instance fallback, used only when Redis isn't configured
function memoryLimiter(max: number, windowMs: number): Limiter {
  const hits = new Map<string, { count: number; reset: number }>();
  setInterval(() => {
    const now = Date.now();
    for (const [k, v] of hits) if (v.reset < now) hits.delete(k);
  }, 60_000).unref();

  return {
    async limit(key) {
      const now = Date.now();
      let h = hits.get(key);
      if (!h || h.reset < now) {
        h = { count: 0, reset: now + windowMs };
        hits.set(key, h);
      }
      h.count++;
      return { success: h.count <= max, limit: max, remaining: Math.max(0, max - h.count), reset: h.reset };
    },
  };
}

function make(name: string, max: number, windowSec: number): Limiter {
  if (!redis) return memoryLimiter(max, windowSec * 1000);
  return new Ratelimit({
    redis,
    limiter: Ratelimit.slidingWindow(max, `${windowSec} s`),
    prefix: `${PREFIX}:rl:${name}`,
    ephemeralCache: new Map(), // remembers blocked IPs in memory, so a flood doesn't cost Redis commands
    analytics: false,          // analytics writes extra commands
  });
}
